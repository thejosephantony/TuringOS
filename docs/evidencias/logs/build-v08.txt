mkdir -p build
nasm -f elf32 boot/boot.asm -o build/boot.o
gcc -m32 -Iinclude -ffreestanding -fno-builtin -fno-pic -fno-pie -fno-stack-protector -nostdlib -Wall -Wextra -c kernel/kernel.c -o build/kernel.o
gcc -m32 -Iinclude -ffreestanding -fno-builtin -fno-pic -fno-pie -fno-stack-protector -nostdlib -Wall -Wextra -c kernel/vga.c -o build/vga.o
gcc -m32 -Iinclude -ffreestanding -fno-builtin -fno-pic -fno-pie -fno-stack-protector -nostdlib -Wall -Wextra -c kernel/terminal.c -o build/terminal.o
gcc -m32 -Iinclude -ffreestanding -fno-builtin -fno-pic -fno-pie -fno-stack-protector -nostdlib -Wall -Wextra -c kernel/shell.c -o build/shell.o
gcc -m32 -Iinclude -ffreestanding -fno-builtin -fno-pic -fno-pie -fno-stack-protector -nostdlib -Wall -Wextra -c kernel/keyboard.c -o build/keyboard.o
nasm -f elf32 kernel/interrupt.asm -o build/interrupt.o
gcc -m32 -Iinclude -ffreestanding -fno-builtin -fno-pic -fno-pie -fno-stack-protector -nostdlib -Wall -Wextra -c kernel/idt.c -o build/idt.o
gcc -m32 -Iinclude -ffreestanding -fno-builtin -fno-pic -fno-pie -fno-stack-protector -nostdlib -Wall -Wextra -c kernel/isr.c -o build/isr.o
gcc -m32 -Iinclude -ffreestanding -fno-builtin -fno-pic -fno-pie -fno-stack-protector -nostdlib -Wall -Wextra -c kernel/pic.c -o build/pic.o
gcc -m32 -Iinclude -ffreestanding -fno-builtin -fno-pic -fno-pie -fno-stack-protector -nostdlib -Wall -Wextra -c kernel/timer.c -o build/timer.o
gcc -m32 -Iinclude -ffreestanding -fno-builtin -fno-pic -fno-pie -fno-stack-protector -nostdlib -Wall -Wextra -c kernel/klib.c -o build/klib.o
ld -m elf_i386 -T linker.ld -nostdlib -o build/turingos.bin build/boot.o build/kernel.o build/vga.o build/terminal.o build/shell.o build/keyboard.o build/interrupt.o build/idt.o build/isr.o build/pic.o build/timer.o build/klib.o
cp build/turingos.bin iso/boot/turingos.bin
